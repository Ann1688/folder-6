檢視 Google Apps Script 後端接收範例碼

請將以下程式碼貼入 GAS 編輯器（`Code.gs`），並部署為 Web App：

function doPost(e) {
  try {
    // 解包 JSON 資料 (因 no-cors，資料在 e.postData.contents)
    var data = JSON.parse(e.postData.contents);
    
    // 取得試算表 (範例寫入第一個 Sheet)
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // 寫入欄位：團隊編號, UUID, 姓名, 緯度, 經度, 時間
    sheet.appendRow([
      data.teamCode,
      data.uuid,
      data.name || '',
      data.latitude,
      data.longitude,
      data.timestamp
    ]);
    
    return ContentService.createTextOutput("Success");
  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString());
  }
}
